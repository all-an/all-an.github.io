Constant-Time Secret Comparison

Standard String.equals stops at the first mismatched byte, allowing attackers to measure nanosecond response differences and reconstruct HMAC signatures or API keys byte-by-byte. Always compare secrets with MessageDigest.isEqual.

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;

public final class ConstantTimeVerifier {

    // VULNERABLE: Early-exit byte comparison leaks how many leading bytes matched
    public static boolean insecureEquals(String a, String b) {
        return a.equals(b);
    }

    // HARDENED: Examines all bytes in constant time regardless of where a mismatch occurs
    public static boolean secureEquals(String expectedSecret, String providedSecret) {
        if (expectedSecret == null || providedSecret == null) {
            return false;
        }
        byte[] expectedBytes = expectedSecret.getBytes(StandardCharsets.UTF_8);
        byte[] providedBytes = providedSecret.getBytes(StandardCharsets.UTF_8);
        return MessageDigest.isEqual(expectedBytes, providedBytes);
    }
}
