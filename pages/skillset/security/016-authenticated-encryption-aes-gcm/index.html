Authenticated Encryption at Rest (AES-256-GCM)

Never use AES/ECB (which leaks patterns) or unauthenticated AES/CBC (vulnerable to padding-oracle attacks). Use AES/GCM/NoPadding with a 256-bit key, a unique 12-byte SecureRandom IV per encryption, and a 128-bit authentication tag that detects any ciphertext tampering.

public final class AesGcmFieldEncryptor {
    private static final String ALGO = "AES/GCM/NoPadding";
    private static final int IV_BYTES = 12;     // 96-bit NIST recommended nonce
    private static final int TAG_BITS = 128;    // 128-bit authentication tag
    private final SecretKey key;
    private final SecureRandom secureRandom = new SecureRandom();

    public AesGcmFieldEncryptor(SecretKey key) { this.key = key; }

    public byte[] encrypt(byte[] plaintext, byte[] associatedData) throws GeneralSecurityException {
        byte[] iv = new byte[IV_BYTES];
        secureRandom.nextBytes(iv);

        Cipher cipher = Cipher.getInstance(ALGO);
        cipher.init(Cipher.ENCRYPT_MODE, key, new GCMParameterSpec(TAG_BITS, iv));
        if (associatedData != null) cipher.updateAAD(associatedData);
        byte[] ciphertext = cipher.doFinal(plaintext);

        return ByteBuffer.allocate(iv.length + ciphertext.length).put(iv).put(ciphertext).array();
    }
}
