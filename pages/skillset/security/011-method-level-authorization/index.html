Method-Level Authorization & Least Privilege

Relying only on URL path matchers in a web filter leaves service methods exposed if called from another controller, GraphQL resolver, or scheduled task. Apply @EnableMethodSecurity and @PreAuthorize directly on service methods for defense in depth.

@Configuration
@EnableMethodSecurity
public class MethodSecurityConfig {}

@Service
public class AccountAdminService {

    // Verifies role AND custom SpEL ownership/tenant rule before method execution
    @PreAuthorize("hasRole('COMPLIANCE_OFFICER') and #tenantId == authentication.token.claims['tenant_id']")
    public void freezeAccount(String tenantId, UUID accountId, String reason) {
        accountRepository.updateStatus(tenantId, accountId, AccountStatus.FROZEN);
    }

    // PostAuthorize can also verify the returned object belongs to the caller
    @PostAuthorize("returnObject.ownerId == authentication.name or hasRole('ADMIN')")
    public AccountReport loadReport(UUID reportId) {
        return reportRepository.findById(reportId).orElseThrow();
    }
}
