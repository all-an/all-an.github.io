Strict CORS & Anti-CSRF Cookie Hardening

Reflecting arbitrary Origin headers with Access-Control-Allow-Credentials: true lets malicious websites read authenticated API responses. Restrict CORS to an explicit origin allowlist, and when using cookies, set HttpOnly, Secure, and SameSite=Strict.

@Bean
public CorsConfigurationSource corsConfigurationSource() {
    CorsConfiguration config = new CorsConfiguration();
    config.setAllowedOrigins(List.of("https://app.example.com")); // Never "*" with credentials
    config.setAllowedMethods(List.of("GET", "POST", "PUT", "DELETE"));
    config.setAllowedHeaders(List.of("Authorization", "Content-Type"));
    config.setAllowCredentials(true);
    config.setMaxAge(3600L);

    UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
    source.registerCorsConfiguration("/api/**", config);
    return source;
}

// Hardened session/refresh cookie construction:
ResponseCookie cookie = ResponseCookie.from("__Host-refresh_token", tokenValue)
    .httpOnly(true)       // Inaccessible to document.cookie (XSS defense)
    .secure(true)         // Sent over HTTPS only
    .path("/")
    .sameSite("Strict")   // Blocks cross-site request forgery (CSRF)
    .maxAge(Duration.ofDays(7))
    .build();
