SQL Injection & Parameterized Queries

Concatenating untrusted input into SQL or JPQL strings lets attackers inject ' OR '1'='1 or UNION SELECT payloads to dump or destroy tables. Always use PreparedStatement or named @Param bindings, and validate dynamic sort columns against a strict allowlist.

// VULNERABLE: String concatenation allows SQL injection
String unsafeSql = "SELECT * FROM users WHERE email = '" + email + "'";
jdbcTemplate.queryForList(unsafeSql);

// HARDENED (JDBC): Parameterized PreparedStatement binds input as a literal value
String safeSql = "SELECT id, email, role FROM users WHERE email = ?";
User user = jdbcTemplate.queryForObject(
    safeSql,
    (rs, rowNum) -> new User(rs.getUUID("id"), rs.getString("email"), rs.getString("role")),
    email
);

// HARDENED (Spring Data JPA): Named @Param binding + allowlisted sort column
@Query("SELECT o FROM Order o WHERE o.customerId = :customerId AND o.status = :status")
List<Order> findOrders(@Param("customerId") String customerId, @Param("status") OrderStatus status);
