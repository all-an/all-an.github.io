Multi-Factor Auth (TOTP) & Step-Up Auth

Stolen credentials or session tokens should not be enough to drain funds or export customer databases. Require Time-Based One-Time Passwords (RFC 6238 TOTP / WebAuthn) at login and enforce step-up verification before high-privilege operations.

@Service
public class PayoutService {

    private final TotpVerifier totpVerifier;
    private final AuditLogger auditLogger;

    @PreAuthorize("hasAuthority('SCOPE_payout:write')")
    public PayoutReceipt executeHighValuePayout(
            UserPrincipal principal,
            BigDecimal amount,
            String totpCode) {

        // Step-up authentication required for payouts above threshold
        if (amount.compareTo(new BigDecimal("1000.00")) > 0) {
            if (!totpVerifier.verifyCode(principal.userId(), totpCode)) {
                auditLogger.logSecurityAlert("STEP_UP_MFA_FAILED", principal.userId());
                throw new AccessDeniedException("Valid MFA code required for high-value payout");
            }
        }
        return processTransfer(principal.userId(), amount);
    }
}
