Dependency injection

Angular’s hierarchical injector supplies dependencies to classes rather than having them build their own. The inject() function is the modern way to request one.

import { Component, inject } from '@angular/core';
import { CartService } from './cart.service';

@Component({ selector: 'app-cart', standalone: true, template: '{{ cart.count }}' })
export class CartComponent {
  // Angular resolves and provides the singleton instance
  cart = inject(CartService);
}
