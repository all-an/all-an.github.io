Router

The Angular Router maps URL paths to components and renders them into a <router-outlet>. routerLink navigates without full page reloads.

import { Routes } from '@angular/router';

export const routes: Routes = [
  { path: '', component: HomeComponent },
  { path: 'users/:id', component: UserComponent }, // route parameter
  { path: '**', component: NotFoundComponent },    // wildcard
];
// Template:  <a routerLink="/users/1">User 1</a>  <router-outlet />
