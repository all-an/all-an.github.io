Pipe

A pipe transforms a value for display with the | operator. Angular ships pipes like date, currency and async, and you can write your own with @Pipe.

import { Pipe, PipeTransform } from '@angular/core';

@Pipe({ name: 'exclaim', standalone: true })
export class ExclaimPipe implements PipeTransform {
  transform(value: string): string {
    return value + '!';
  }
}
// Usage in a template:  {{ 'hello' | exclaim }}  ->  hello!
