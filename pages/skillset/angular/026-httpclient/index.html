HttpClient

HttpClient makes HTTP requests that return Observables. Register it once with provideHttpClient(), then inject it wherever you need it.

import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';

@Injectable({ providedIn: 'root' })
export class UserApi {
  private http = inject(HttpClient);

  getUsers() { return this.http.get<User[]>('/api/users'); }
  create(u: User) { return this.http.post<User>('/api/users', u); }
}
