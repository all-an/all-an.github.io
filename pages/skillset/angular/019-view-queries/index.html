View queries

View queries grab references to child elements, directives or components from a template. The signal-based viewChild / contentChild (v17.3+) return reactive signals.

import { Component, viewChild, ElementRef, AfterViewInit } from '@angular/core';

@Component({
  selector: 'app-search',
  standalone: true,
  template: '<input #box />',
})
export class SearchComponent implements AfterViewInit {
  box = viewChild<ElementRef<HTMLInputElement>>('box');

  ngAfterViewInit() {
    this.box()?.nativeElement.focus(); // available once the view exists
  }
}
