Form validation

Validators attach rules to controls; failing rules populate errors and mark the form invalid. Angular provides built-in validators and supports custom ones.

import { FormControl, Validators } from '@angular/forms';

const email = new FormControl('', [
  Validators.required,
  Validators.email,
]);

// Custom validator: value must not be blank/whitespace
function notBlank(c: FormControl) {
  return c.value?.trim() ? null : { blank: true };
}
