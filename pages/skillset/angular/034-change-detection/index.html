Change detection

Change detection syncs component state to the DOM. The OnPush strategy only re-checks a component when its inputs change or an event fires — a key performance lever.

import { Component, ChangeDetectionStrategy, input } from '@angular/core';

@Component({
  selector: 'app-row',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush, // check only on input change
  template: '{{ item().name }}',
})
export class RowComponent {
  item = input.required<Item>();
}
