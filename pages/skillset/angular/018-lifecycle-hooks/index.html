Lifecycle hooks

Angular calls lifecycle hooks at key moments — creation, change, destruction. ngOnInit initializes; ngOnDestroy cleans up subscriptions and timers.

import { Component, OnInit, OnDestroy } from '@angular/core';

@Component({ selector: 'app-timer', standalone: true, template: '{{ ticks }}' })
export class TimerComponent implements OnInit, OnDestroy {
  ticks = 0;
  private id?: ReturnType<typeof setInterval>;

  ngOnInit() { this.id = setInterval(() => this.ticks++, 1000); }
  ngOnDestroy() { clearInterval(this.id); } // avoid leaks
}
