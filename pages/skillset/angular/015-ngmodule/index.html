NgModule

An @NgModule groups components, directives and pipes and wires up providers. It was the primary organizing unit before standalone components; you still meet it in older codebases.

import { NgModule } from '@angular/core';
import { CommonModule } from '@angular/common';
import { UserComponent } from './user.component';

@NgModule({
  declarations: [UserComponent], // components owned by this module
  imports: [CommonModule],       // modules it depends on
  exports: [UserComponent],      // what it exposes to importers
})
export class UserModule {}
