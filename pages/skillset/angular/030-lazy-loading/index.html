Lazy loading

Lazy loading defers a route’s code until the user navigates to it, shrinking the initial bundle. Use loadComponent for a component or loadChildren for a set of routes.

export const routes: Routes = [
  {
    path: 'admin',
    // Loaded on demand, split into its own chunk
    loadComponent: () =>
      import('./admin/admin.component').then(m => m.AdminComponent),
  },
];
