Input & Output

Components communicate through inputs (data in) and outputs (events out). The signal-based input() and output() functions (v17.1+) are the modern API.

import { Component, input, output } from '@angular/core';

@Component({ selector: 'app-user', standalone: true, template: '...' })
export class UserComponent {
  user = input.required<User>();   // parent -> child data
  deleted = output<number>();      // child -> parent event

  remove() {
    this.deleted.emit(this.user().id); // notify the parent
  }
}
