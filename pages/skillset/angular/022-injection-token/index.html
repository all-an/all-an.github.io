Injection token

An InjectionToken is a typed key for injecting values that are not classes — config objects, strings, or swappable implementations.

import { InjectionToken, inject } from '@angular/core';

export interface AppConfig { apiUrl: string; }
export const APP_CONFIG = new InjectionToken<AppConfig>('app.config');

// Provide it at bootstrap:
// providers: [{ provide: APP_CONFIG, useValue: { apiUrl: '/api' } }]

const config = inject(APP_CONFIG); // typed as AppConfig
