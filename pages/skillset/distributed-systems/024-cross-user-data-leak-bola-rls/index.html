Cross-User Data Access, BOLA/IDOR & PostgreSQL Row-Level Security (RLS)

In multi-user and multi-tenant SaaS applications, a single forgotten WHERE tenant_id = :tenantId clause exposes another customer’s private invoices or medical records. Enforce tenant/owner scoping in every repository method and configure PostgreSQL Row-Level Security (RLS) with transaction-local session variables (SET LOCAL) so the database itself refuses cross-tenant reads.

// LAYER 1 (Application Repository): Never expose unscoped findById(id) on tenant resources
public interface InvoiceRepository extends JpaRepository<Invoice, UUID> {
    @Query("SELECT i FROM Invoice i WHERE i.id = :id AND i.tenantId = :tenantId")
    Optional<Invoice> findByIdAndTenantId(@Param("id") UUID id, @Param("tenantId") UUID tenantId);
}

// LAYER 2 (Database Defense-in-Depth): Bind tenantId to PostgreSQL transaction-local context for RLS
// SQL Policy: CREATE POLICY tenant_isolation ON invoices USING (tenant_id = current_setting('app.tenant_id')::uuid);
@Aspect
@Component
public class TenantRlsAspect {
    @Before("@annotation(org.springframework.transaction.annotation.Transactional)")
    public void bindTenantToDatabaseTransaction() {
        UUID tenantId = TenantContext.getCurrentTenantId();
        // SET LOCAL scopes the setting strictly to the current SQL transaction (safe with HikariCP pooling!)
        jdbcTemplate.execute("SELECT set_config('app.tenant_id', '" + tenantId + "', true)");
    }
}
