Out-of-Order Events, Partition Keys & State Machine Guards

If events for the same order are published to Kafka without a partition key, OrderCancelled can land on Partition 2 and be consumed before OrderCreated on Partition 0. Always use the aggregate ID (orderId) as the Kafka message key so all events for an entity stay ordered in one partition, and guard state transitions with monotonic version numbers.

// 1. PRODUCER: Always set aggregateId as the Kafka partition key to preserve per-entity order
public void publishOrderEvent(OrderEvent event) {
    // Same orderId always hashes to the same Kafka partition
    kafkaTemplate.send("orders.lifecycle", event.orderId().toString(), event);
}

// 2. CONSUMER: Enforce state machine transition rules and monotonic event sequence numbers
@Transactional
public void applyOrderEvent(OrderEvent incoming) {
    OrderProjection proj = projectionRepo.findByIdForUpdate(incoming.orderId())
        .orElseGet(() -> new OrderProjection(incoming.orderId(), OrderState.NONE, 0L));

    // Ignore stale/out-of-order events that have a lower or equal sequence number
    if (incoming.sequenceNumber() <= proj.getLastSequenceNumber()) {
        log.warn("Ignoring stale event seq={} (current={})", incoming.sequenceNumber(), proj.getLastSequenceNumber());
        return;
    }

    // Guard invalid state transitions (e.g. cannot transition SHIPPED -> CANCELLED)
    if (!proj.getState().canTransitionTo(incoming.targetState())) {
        throw new IllegalStateTransitionException(proj.getState(), incoming.targetState());
    }

    proj.setState(incoming.targetState());
    proj.setLastSequenceNumber(incoming.sequenceNumber());
    projectionRepo.save(proj);
}
