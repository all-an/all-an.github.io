Last-Item Inventory Overselling & Pessimistic Locking

When two users simultaneously buy the last item in stock, a plain findById reads stock = 1 in both transactions before either writes stock = 0, resulting in two confirmed orders for one physical item. Use SELECT ... FOR UPDATE (LockModeType.PESSIMISTIC_WRITE) with a lock timeout so the second transaction waits until the first commits and then sees stock = 0.

// BROKEN: Check-Then-Act race condition — two threads both read stock=1 and oversell
@Transactional
public Order checkoutBroken(UUID productId, int qty) {
    Product p = productRepo.findById(productId).orElseThrow();
    if (p.getStock() < qty) {
        throw new OutOfStockException();
    }
    p.setStock(p.getStock() - qty); // Both threads write 1 - 1 = 0!
    return orderRepo.save(new Order(productId, qty));
}

// PRODUCTION-SAFE: Pessimistic row-level exclusive lock (SELECT ... FOR UPDATE)
public interface ProductRepository extends JpaRepository<Product, UUID> {
    @Lock(LockModeType.PESSIMISTIC_WRITE)
    @QueryHints(@QueryHint(name = "jakarta.persistence.lock.timeout", value = "3000"))
    @Query("SELECT p FROM Product p WHERE p.id = :id")
    Optional<Product> findByIdForUpdate(@Param("id") UUID id);
}

@Transactional
public Order checkoutSafe(UUID productId, int qty) {
    // Second buyer blocks here until the first transaction commits, then sees stock = 0
    Product p = productRepo.findByIdForUpdate(productId).orElseThrow();
    if (p.getStock() < qty) {
        throw new OutOfStockException("Sold out while waiting for lock");
    }
    p.setStock(p.getStock() - qty);
    return orderRepo.save(new Order(productId, qty));
}
