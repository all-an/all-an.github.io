Optimistic Locking (@Version / CAS) & Atomic SQL Decrement

Holding pessimistic locks across long business workflows reduces throughput. Instead, use JPA @Version for optimistic compare-and-swap (CAS) detection, or push the check and decrement into a single atomic SQL UPDATE ... WHERE stock >= :qty statement that never allows negative stock.

// OPTION 1: JPA Optimistic Locking (@Version column automatically checked on UPDATE)
@Entity
public class Product {
    @Id private UUID id;
    private int stock;
    @Version private long version; // UPDATE products SET stock=?, version=version+1 WHERE id=? AND version=?
}

// OPTION 2 (FASTEST FOR HOT INVENTORY): Single atomic conditional UPDATE in PostgreSQL
public interface ProductRepository extends JpaRepository<Product, UUID> {
    @Modifying
    @Query("UPDATE Product p SET p.stock = p.stock - :qty WHERE p.id = :id AND p.stock >= :qty")
    int tryDecrementStock(@Param("id") UUID id, @Param("qty") int qty);
}

@Transactional
public Order checkoutAtomic(UUID productId, int qty) {
    // Database executes the check and decrement atomically on the row lock
    int updatedRows = productRepo.tryDecrementStock(productId, qty);
    if (updatedRows == 0) {
        throw new OutOfStockException("Insufficient stock for product " + productId);
    }
    return orderRepo.save(new Order(productId, qty));
}
