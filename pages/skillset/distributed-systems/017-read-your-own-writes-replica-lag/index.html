Read-Your-Own-Writes Anomaly & Read-Replica Lag Routing

When writes go to a primary database and reads go to asynchronous read replicas, a user who updates their address and immediately reloads the page within 100ms often sees their old address—making them think the save failed. Pin a user or session to the primary database for a brief window after any write (or compare PostgreSQL WAL LSNs) to guarantee Read-Your-Own-Writes consistency.

// PRODUCTION-SAFE: Route reads to Primary if the current user performed a write within the last 5 seconds
public class ReplicaAwareRoutingDataSource extends AbstractRoutingDataSource {
    @Override
    protected Object determineCurrentLookupKey() {
        if (TransactionSynchronizationManager.isCurrentTransactionReadOnly()
                && !RecentWriteTracker.hasRecentWriteForCurrentUser()) {
            return "REPLICA";
        }
        return "PRIMARY";
    }
}

public final class RecentWriteTracker {
    private static final Cache<UUID, Boolean> RECENT_WRITERS = Caffeine.newBuilder()
        .expireAfterWrite(Duration.ofSeconds(5)) // Covers typical async replication lag window
        .build();

    public static void markUserWrote(UUID userId) {
        RECENT_WRITERS.put(userId, Boolean.TRUE);
    }

    public static boolean hasRecentWriteForCurrentUser() {
        UUID userId = SecurityContextUtil.currentUserIdOrNull();
        return userId != null && RECENT_WRITERS.getIfPresent(userId) != null;
    }
}
