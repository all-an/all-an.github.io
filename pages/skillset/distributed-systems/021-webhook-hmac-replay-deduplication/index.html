Webhook Spoofing, Replay Attacks & Event Deduplication

Webhook receivers (Stripe, Adyen, GitHub) are public endpoints callable by anyone on the internet. Always verify the HMAC-SHA256 signature computed over timestamp + "." + rawBody using constant-time comparison (MessageDigest.isEqual), reject timestamps older than 5 minutes to prevent replay attacks, and deduplicate by the provider event ID.

// PRODUCTION-SAFE: Constant-time HMAC verification + 5-minute replay window + event ID deduplication
public void handleWebhook(String signatureHeader, String timestampHeader, byte[] rawBody) {
    long timestampSec = Long.parseLong(timestampHeader);
    long nowSec       = Instant.now().getEpochSecond();
    if (Math.abs(nowSec - timestampSec) > 300) {
        throw new SecurityException("Webhook timestamp outside 5-minute replay tolerance");
    }

    byte[] signedPayload = (timestampHeader + "." + new String(rawBody, StandardCharsets.UTF_8))
        .getBytes(StandardCharsets.UTF_8);
    byte[] expectedMac = computeHmacSha256(webhookSecretKey, signedPayload);
    byte[] providedMac = HexFormat.of().parseHex(signatureHeader);

    // Constant-time comparison prevents byte-by-byte timing side-channel attacks
    if (!MessageDigest.isEqual(expectedMac, providedMac)) {
        throw new SecurityException("Invalid webhook HMAC signature");
    }

    WebhookEvent evt = objectMapper.readValue(rawBody, WebhookEvent.class);
    Boolean isFirstSeen = redisTemplate.opsForValue()
        .setIfAbsent("webhook:dedup:" + evt.id(), "1", Duration.ofHours(24));
    if (!Boolean.TRUE.equals(isFirstSeen)) {
        return; // Already processed this webhook event ID
    }
    processVerifiedWebhook(evt);
}
