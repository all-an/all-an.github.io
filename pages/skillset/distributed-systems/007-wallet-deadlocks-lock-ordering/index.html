Concurrent Wallet Transfers & Deterministic Lock Ordering

When Alice transfers $10 to Bob at the exact same millisecond that Bob transfers $20 to Alice, locking fromAccount then toAccount causes Thread 1 to lock Alice and wait for Bob while Thread 2 locks Bob and waits for Alice—a classic database deadlock. Always acquire locks in a deterministic global order (e.g., ascending UUID/ID).

// BROKEN: Locking in parameter order (fromId, then toId) deadlocks on A->B + B->A
@Transactional
public void transferDeadlockProne(UUID fromId, UUID toId, long cents) {
    Account from = accountRepo.findByIdForUpdate(fromId).orElseThrow(); // T1 locks A, T2 locks B
    Account to   = accountRepo.findByIdForUpdate(toId).orElseThrow();   // T1 waits B, T2 waits A -> DEADLOCK!
    from.debit(cents);
    to.credit(cents);
}

// PRODUCTION-SAFE: Acquire locks in deterministic ascending ID order
@Transactional(rollbackFor = Exception.class)
public void transferDeadlockFree(UUID fromId, UUID toId, long cents) {
    if (fromId.equals(toId)) {
        throw new IllegalArgumentException("Cannot transfer to the same account");
    }
    UUID firstLockId  = fromId.compareTo(toId) < 0 ? fromId : toId;
    UUID secondLockId = fromId.compareTo(toId) < 0 ? toId : fromId;

    // Both T1 (A->B) and T2 (B->A) lock Account A first, then Account B — zero deadlock cycle!
    Account first  = accountRepo.findByIdForUpdate(firstLockId).orElseThrow();
    Account second = accountRepo.findByIdForUpdate(secondLockId).orElseThrow();

    Account from = fromId.equals(firstLockId) ? first : second;
    Account to   = toId.equals(firstLockId)   ? first : second;

    from.debit(cents);
    to.credit(cents);
}
