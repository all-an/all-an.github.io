Path variables & params

@PathVariable binds a URI template segment; @RequestParam binds a query-string or form parameter, with optional defaults.

// GET /search/books?limit=20
@GetMapping("/search/{category}")
public List<Item> search(
        @PathVariable String category,
        @RequestParam(defaultValue = "10") int limit) {
    return service.search(category, limit);
}
