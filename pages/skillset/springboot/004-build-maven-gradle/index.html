Build: Maven / Gradle

Spring Boot projects build with Maven or Gradle. The Spring Boot plugin packages your code and dependencies into a single executable “fat” jar.

<!-- pom.xml — inherit sensible defaults from the Boot parent -->
<parent>
    <groupId>org.springframework.boot</groupId>
    <artifactId>spring-boot-starter-parent</artifactId>
    <version>3.3.0</version>
</parent>

<build>
    <plugins>
        <plugin>
            <groupId>org.springframework.boot</groupId>
            <artifactId>spring-boot-maven-plugin</artifactId>
        </plugin>
    </plugins>
</build>
