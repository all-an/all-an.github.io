@SpringBootApplication

@SpringBootApplication is a meta-annotation combining @Configuration, @EnableAutoConfiguration and @ComponentScan. It marks the entry point and scans its package for components.

@SpringBootApplication // = @Configuration + @EnableAutoConfiguration + @ComponentScan
public class StoreApplication {
    public static void main(String[] args) {
        // Boots the context, starts the embedded server
        SpringApplication.run(StoreApplication.class, args);
    }
}
