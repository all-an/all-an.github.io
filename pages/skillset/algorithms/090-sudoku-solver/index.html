Sudoku solver

Hard Backtracking

Fill a 9 × 9 Sudoku grid so that every row, every column and every 3 × 3 box contains the digits 1–9 exactly once. Backtracking solves it: try a digit in the next empty cell, recurse, and undo it if the rest cannot be completed.

Try it

↓

Approach

Find the next empty cell. Try each digit 1–9; a digit is legal if it does not already appear in that cell's row, column or 3 × 3 box. Place a legal digit and recurse to solve the rest of the board. If the recursion succeeds, you are done; if not, erase the digit (backtrack) and try the next one. If no digit works, return false to the caller, which then revises its own earlier choice.

The box containing cell (r, c) starts at (3·⌊r/3⌋, 3·⌊c/3⌋). Keeping per-row, per-column and per-box "digit used" sets makes each legality check O(1), and choosing the empty cell with the fewest candidates first speeds things up a great deal.

Java

// Backtracking solver: board holds digits 1-9, and '.' for empty cells. Solves in place.
static boolean solveSudoku(char[][] board) {
    for (int r = 0; r < 9; r++) {
        for (int c = 0; c < 9; c++) {
            if (board[r][c] != '.') continue;
            for (char digit = '1'; digit <= '9'; digit++) {
                if (isLegal(board, r, c, digit)) {
                    board[r][c] = digit;                 // try it
                    if (solveSudoku(board)) return true;
                    board[r][c] = '.';                   // backtrack
                }
            }
            return false; // no digit fits this cell: an earlier choice was wrong
        }
    }
    return true; // no empty cell left
}

// A digit is legal if it is absent from the cell's row, column and 3x3 box.
static boolean isLegal(char[][] board, int r, int c, char digit) {
    for (int i = 0; i < 9; i++) {
        if (board[r][i] == digit || board[i][c] == digit) return false;
        int boxRow = 3 * (r / 3) + i / 3;
        int boxColumn = 3 * (c / 3) + i % 3;
        if (board[boxRow][boxColumn] == digit) return false;
    }
    return true;
}

JavaScript

// Backtracking solver: board holds digit characters '1'-'9', and '.' for empty cells. Solves in place.
function solveSudoku(board) {
  for (let r = 0; r < 9; r++) {
    for (let c = 0; c < 9; c++) {
      if (board[r][c] !== '.') continue;
      for (let digit = 1; digit <= 9; digit++) {
        const candidate = String(digit);
        if (isLegal(board, r, c, candidate)) {
          board[r][c] = candidate;               // try it
          if (solveSudoku(board)) return true;
          board[r][c] = '.';                     // backtrack
        }
      }
      return false; // no digit fits this cell: an earlier choice was wrong
    }
  }
  return true; // no empty cell left
}

// A digit is legal if it is absent from the cell's row, column and 3x3 box.
function isLegal(board, r, c, digit) {
  for (let i = 0; i < 9; i++) {
    if (board[r][i] === digit || board[i][c] === digit) return false;
    const boxRow = 3 * Math.floor(r / 3) + Math.floor(i / 3);
    const boxColumn = 3 * Math.floor(c / 3) + (i % 3);
    if (board[boxRow][boxColumn] === digit) return false;
  }
  return true;
}

Complexity

  • Time O(9^m) — m empty cells — a loose bound, since pruning removes almost every branch
  • Space O(m) — recursion depth is at most the number of empty cells

Real solvers add constraint propagation (fill cells that have only one candidate) before guessing. The exhaustive backtracking above is what interviews expect, and it solves ordinary puzzles in milliseconds.
