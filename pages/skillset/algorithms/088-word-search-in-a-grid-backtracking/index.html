Word search in a grid (backtracking)

Hard Backtracking

Given a grid of letters and a word, decide whether the word can be traced through adjacent cells (up, down, left, right) without using any cell twice. It is backtracking on a grid: try a path, and undo it if it dead-ends.

Try it

↓

Approach

Try every cell as a starting point. From a cell whose letter matches the next needed letter of the word, mark the cell as visited, then explore its four neighbours for the following letter, recursively. If any neighbour finishes the word, report success. If none does, backtrack: un-mark the cell (restore its letter) so other paths may use it, and return false.

Reuse the grid itself as the visited marker by temporarily overwriting the cell with a sentinel such as '#' — no separate visited array is needed. Prune early: stop the moment a letter fails to match.

Java

// Try every cell as a start; DFS with backtracking.
static boolean exist(char[][] grid, String word) {
    for (int r = 0; r < grid.length; r++) {
        for (int c = 0; c < grid[r].length; c++) {
            if (search(grid, word, r, c, 0)) return true;
        }
    }
    return false;
}

static boolean search(char[][] grid, String word, int r, int c, int index) {
    if (index == word.length()) return true; // every letter matched
    if (r < 0 || c < 0 || r >= grid.length || c >= grid[r].length || grid[r][c] != word.charAt(index)) {
        return false;
    }
    char saved = grid[r][c];
    grid[r][c] = '#'; // mark visited so this path cannot reuse the cell
    boolean found = search(grid, word, r + 1, c, index + 1)
                 || search(grid, word, r - 1, c, index + 1)
                 || search(grid, word, r, c + 1, index + 1)
                 || search(grid, word, r, c - 1, index + 1);
    grid[r][c] = saved; // backtrack: free the cell for other paths
    return found;
}

JavaScript

// Try every cell as a start; DFS with backtracking.
function exist(grid, word) {
  for (let r = 0; r < grid.length; r++) {
    for (let c = 0; c < grid[r].length; c++) {
      if (search(grid, word, r, c, 0)) return true;
    }
  }
  return false;
}

function search(grid, word, r, c, index) {
  if (index === word.length) return true; // every letter matched
  if (r < 0 || c < 0 || r >= grid.length || c >= grid[r].length || grid[r][c] !== word[index]) {
    return false;
  }
  const saved = grid[r][c];
  grid[r][c] = '#'; // mark visited so this path cannot reuse the cell
  const found = search(grid, word, r + 1, c, index + 1)
             || search(grid, word, r - 1, c, index + 1)
             || search(grid, word, r, c + 1, index + 1)
             || search(grid, word, r, c - 1, index + 1);
  grid[r][c] = saved; // backtrack: free the cell for other paths
  return found;
}

Complexity

  • Time O(R × C × 3ᴸ) — L is the word length; each step has at most 3 new directions
  • Space O(L) — recursion depth equals the word length

To search for many words at once ("Word Search II"), build a trie of the words and walk it during the DFS so every path is checked against all words simultaneously.
